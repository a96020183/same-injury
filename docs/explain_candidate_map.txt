FROM lightning.metastore.l1.unseen3001_records.incident i
JOIN lightning.metastore.l1.unseen3001_records.site s ON i.site_code = s.site_code
JOIN unseen3001_public.severe_injury_reports r ON regexp_extract(r.Address1, '^ *([0-9]+)', 1) = s.street_number AND (r.Zip = s.zip OR upper(r.City) = upper(s.city))

LEFT JOIN unseen3001_public.related_activity a ON a.rel_act_nr = r.UPA AND a.rel_type = 'R'
LEFT JOIN unseen3001_public.inspection n ON n.activity_nr = a.activity_nr
WHERE i.event_date >= '2024-01-01' AND i.narrative rlike '(?i)amputat'
